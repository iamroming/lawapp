    const { data: profile } = await supabase
      .from("profiles")
      .select("firm_id, role")
      .eq("id", user.id)
      .single();

    const isOwner = ["owner", "partner", "super_admin"].includes(profile.role);

    let query = supabase.from("cases").select("...");
    if (!isOwner) {
      query = query.or(`assigned_to.eq.${user.id},created_by.eq.${user.id}`);
    }

3. REMINDER SYSTEM — FULLY IMPLEMENTED

  Free APIs chosen:
    Email: Resend (100/day free) — resend.com
    SMS: MSG91 (100/month free, India) — msg91.com
    WhatsApp: Interakt (100/month free, India) — interakt.shop
    Push: FCM (unlimited free) — firebase.google.com

  /api/reminders/route.ts (UPDATED)
    POST — Create reminder with firm_id + user_id
           Auto-creates in-app notification
           Sends immediately if reminder_date <= now
    GET  — Fetch reminders (owners see all, employees see own)
    DELETE — Delete reminder with ownership check

    Notification channels supported:
      in_app — Stored in notifications table
      email — Sent via Resend API
      sms — Sent via MSG91 API
      whatsapp — Sent via Interakt API
      push — Placeholder for FCM

  /api/reminders/cron/route.ts (NEW)
    GET/POST — Cron endpoint to process due reminders
    Queries: status=pending AND reminder_date <= now
    Sends notifications for each channel
    Updates reminder status (sent/partial/failed)
    Logs each notification to notification_logs table

    Add to vercel.json:

        "schedule": "* * * * *"
      }]
    }
