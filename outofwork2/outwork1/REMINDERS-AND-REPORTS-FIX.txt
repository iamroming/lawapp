2.1 CURRENT SCHEMA PROBLEM
──────────────────────────
  Most tables do NOT have firm_id. They rely on created_by/assigned_to
  which link to profiles.firm_id. This works but is less secure:

  PROBLEM: If RLS policy is wrong, user sees ALL data across firms
  SOLUTION: Add firm_id to every table for double isolation

2.2 TABLES NEEDING firm_id
───────────────────────────
  Table              Has firm_id?   Has user_id?   Action Needed
  ───────────────────────────────────────────────────────────────
  cases              NO             NO             ADD firm_id, created_by is user_id
  clients            NO             NO             ADD firm_id, created_by is user_id
  documents          NO             NO             ADD firm_id, user_id
  payments           NO             NO             ADD firm_id, received_by is user_id

2.4 RLS POLICIES — Updated with firm_id
─────────────────────────────────────────

  -- Cases: Users see only their firm's cases
  DROP POLICY IF EXISTS "cases_select" ON cases;
  CREATE POLICY "cases_select" ON cases
    FOR SELECT USING (
      firm_id = (
        SELECT firm_id FROM profiles WHERE id = auth.uid()
      )
    );

  DROP POLICY IF EXISTS "cases_insert" ON cases;
  CREATE POLICY "cases_insert" ON cases
    FOR INSERT WITH CHECK (
      firm_id = (
        SELECT firm_id FROM profiles WHERE id = auth.uid()
      )
      AND created_by = auth.uid()
    );

  DROP POLICY IF EXISTS "cases_update" ON cases;
  CREATE POLICY "cases_update" ON cases
    FOR UPDATE USING (
      firm_id = (
        SELECT firm_id FROM profiles WHERE id = auth.uid()
      )
    );

  DROP POLICY IF EXISTS "cases_delete" ON cases;
  CREATE POLICY "cases_delete" ON cases
    FOR DELETE USING (
      firm_id = (
        SELECT firm_id FROM profiles WHERE id = auth.uid()
      )
      AND role IN ('owner', 'partner')
    );

  Code Structure:

    import { NextRequest, NextResponse } from 'next/server';
    import { createClient } from '@/lib/supabase/server';
    import { Resend } from 'resend';

    const resend = new Resend(process.env.RESEND_API_KEY);

    // GET — Fetch reminders for current user's firm
    export async function GET(request: NextRequest) {
      const supabase = await createClient();
      const { data: { user } } = await supabase.auth.getUser();

      if (!user) {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
      }

      // Get user's firm_id
      const { data: profile } = await supabase
        .from('profiles')
        .select('firm_id, role')
        .eq('id', user.id)
        .single();

      if (!profile) {
        return NextResponse.json({ error: 'Profile not found' }, { status: 404 });
      }

      // Fetch reminders for this firm only
      const { data: reminders, error } = await supabase
        .from('scheduled_reminders')
        .select(`
          *,
          case:cases(id, case_number, title),
          client:clients(id, full_name, phone)
        `)
        .eq('firm_id', profile.firm_id)
        .order('reminder_date', { ascending: true });

      if (error) throw error;

      return NextResponse.json({ reminders });
    }

    // POST — Create reminder + send notifications
    export async function POST(request: NextRequest) {
      const supabase = await createClient();
      const { data: { user } } = await supabase.auth.getUser();

      if (!user) {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
      }

      const { data: profile } = await supabase
        .from('profiles')
        .select('firm_id')
        .eq('id', user.id)
        .single();

      const body = await request.json();
      const { title, message, reminder_date, case_id, client_id, channels } = body;

      // Insert reminder with firm_id and user_id
      const { data: reminder, error } = await supabase
        .from('scheduled_reminders')
        .insert({
          firm_id: profile.firm_id,
          user_id: user.id,
          title,
          message,
          reminder_date,
          case_id: case_id || null,
          client_id: client_id || null,
          channels,
          status: 'pending'
        })
        .select()
        .single();

      if (error) throw error;

      // Send immediate in-app notification
      await supabase.from('notifications').insert({
        firm_id: profile.firm_id,
        user_id: user.id,
        type: 'reminder',
        title: `Reminder: ${title}`,
        message,
        data: { reminder_id: reminder.id }
      });

      // If reminder is for now, send immediately
      const reminderTime = new Date(reminder_date);
      const now = new Date();
      if (reminderTime <= now) {
        await sendNotifications(reminder, channels, profile.firm_id);
      }

      return NextResponse.json({ reminder });
    }

    // Helper: Send notifications via all channels
    async function sendNotifications(reminder, channels, firmId) {
      const results = { sent: [], failed: [] };

      for (const channel of channels) {
        try {
          switch (channel) {
            case 'email':
              await sendEmail(reminder);
              results.sent.push(channel);
              break;
            case 'sms':
              await sendSMS(reminder);
              results.sent.push(channel);
              break;
            case 'whatsapp':
              await sendWhatsApp(reminder);
              results.sent.push(channel);
              break;
            case 'in_app':
              results.sent.push(channel);
              break;
            case 'push':
              await sendPushNotification(reminder);
              results.sent.push(channel);
              break;
          }
        } catch (error) {
          results.failed.push(channel);
          console.error(`Failed to send ${channel}:`, error);
        }
      }

      return results;
    }

    // Email sender
    async function sendEmail(reminder) {
      const { data: client } = await supabase
        .from('clients')
        .select('email, full_name')
        .eq('id', reminder.client_id)
        .single();

      if (!client?.email) return;

      await resend.emails.send({
        from: process.env.EMAIL_FROM,
        to: client.email,
        subject: `Reminder: ${reminder.title}`,
        html: `
          <h2>${reminder.title}</h2>
          <p>${reminder.message}</p>
          <p><strong>Date:</strong> ${new Date(reminder.reminder_date).toLocaleString('en-IN')}</p>
          ${reminder.case ? `<p><strong>Case:</strong> ${reminder.case.case_number}</p>` : ''}
        `
      });
    }

    // SMS sender (MSG91)
    async function sendSMS(reminder) {
      const { data: client } = await supabase
        .from('clients')
        .select('phone, full_name')
        .eq('id', reminder.client_id)
        .single();

      if (!client?.phone) return;

      await fetch('https://api.msg91.com/api/v5/flow', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'authkey': process.env.MSG91_API_KEY
        },
        body: JSON.stringify({
          flow_id: 'YOUR_FLOW_ID',
          mobiles: `91${client.phone}`,
          message: `Causly Reminder: ${reminder.title}\n${reminder.message}\nDate: ${new Date(reminder.reminder_date).toLocaleString('en-IN')}`
        })
      });
    }

    // WhatsApp sender (Interakt)
    async function sendWhatsApp(reminder) {
      const { data: client } = await supabase
        .from('clients')
        .select('phone, full_name')
        .eq('id', reminder.client_id)
        .single();

      if (!client?.phone) return;

      await fetch('https://api.interakt.shop/v1/public/message/', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Basic ${Buffer.from(`${process.env.INTERAKT_API_KEY}:${process.env.INTERAKT_TOKEN}`).toString('base64')}`
        },
        body: JSON.stringify({
          countryCode: '+91',
          phoneNumber: client.phone,
          type: 'TEXT',
          message: `Hi ${client.full_name},\n\nReminder: ${reminder.title}\n${reminder.message}\n\nDate: ${new Date(reminder.reminder_date).toLocaleString('en-IN')}\n\n- Causly`
        })
      });
    }

    // Push notification sender (FCM)
    async function sendPushNotification(reminder) {
      // Get user's FCM token from notifications table
      const { data: tokens } = await supabase
        .from('user_devices')
        .select('fcm_token')
        .eq('user_id', reminder.user_id);

      if (!tokens?.length) return;

      for (const { fcm_token } of tokens) {
        await fetch('https://fcm.googleapis.com/v1/projects/' + process.env.FIREBASE_PROJECT_ID + '/messages:send', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${await getAccessToken()}`
          },
          body: JSON.stringify({
            message: {
              token: fcm_token,
              notification: {
                title: reminder.title,
                body: reminder.message
              },
              data: {
                reminder_id: reminder.id,
                case_id: reminder.case_id
              }
            }
          })
        });
      }
    }

3.3 CRON JOB — /api/reminders/cron/route.ts
────────────────────────────────────────────

  File: src/app/api/reminders/cron/route.ts

  Purpose: Check for due reminders every minute and send

  Code:

    import { NextRequest, NextResponse } from 'next/server';
    import { createClient } from '@/lib/supabase/server';

    export async function GET(request: NextRequest) {
      // Verify cron secret to prevent unauthorized access
      const authHeader = request.headers.get('authorization');
      if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
      }

      const supabase = await createClient();

      // Find reminders that are due (status=pending and date <= now)
      const { data: dueReminders, error } = await supabase
        .from('scheduled_reminders')
        .select(`
          *,
          case:cases(id, case_number, title),
          client:clients(id, full_name, phone, email)
        `)
        .eq('status', 'pending')
        .lte('reminder_date', new Date().toISOString());

      if (error) throw error;

      const results = [];

      for (const reminder of dueReminders || []) {
        // Send notifications
        const sendResults = await sendNotifications(reminder, reminder.channels, reminder.firm_id);

        // Update status
        const newStatus = sendResults.failed.length > 0 ? 'partial' : 'sent';
        await supabase
          .from('scheduled_reminders')
          .update({
            status: newStatus,
            sent_channels: sendResults.sent,
            failed_channels: sendResults.failed
          })
          .eq('id', reminder.id);

        // Log each notification
        for (const channel of sendResults.sent) {
          await supabase.from('notification_logs').insert({
            firm_id: reminder.firm_id,
            user_id: reminder.user_id,
            reminder_id: reminder.id,
            channel,
            status: 'sent'
          });
        }

        for (const channel of sendResults.failed) {
          await supabase.from('notification_logs').insert({
            firm_id: reminder.firm_id,
            user_id: reminder.user_id,
            reminder_id: reminder.id,
            channel,
            status: 'failed',
            error_message: 'Failed to send'
          });
        }

        results.push({ reminder_id: reminder.id, status: newStatus });
      }

      return NextResponse.json({ processed: results.length, results });
    }
