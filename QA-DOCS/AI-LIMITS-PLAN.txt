In each route, ADD at the top of POST handler:

  import { checkAiQueryLimit, logAiQuery } from "@/lib/subscription-limits";

  // Check limit
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

  const limitCheck = await checkAiQueryLimit(user.id);
  if (!limitCheck.allowed) {
    return NextResponse.json({
      error: limitCheck.message,
      limitReached: true,
      limit: limitCheck.limit,
      used: limitCheck.used,
      plan: limitCheck.plan,
    }, { status: 429 });
  }

  // ... (existing AI processing logic) ...

  // Log usage after successful processing
  await logAiQuery(user.id, "analyze"); // or "draft", "chat", "research"
